\section{Modelo do Boeing 747}
Segundo \cite{franklin2013,brysonControlSpacecraftAircraft1994},
Reescrevemos as equações de movimento como,
\begin{align}
    \begin{split}
        m(\dot{u} + qw - rv) &= X - mg\sen\theta + \kappa T \cos\theta,\\
        m(\dot{v} + ru - pw) &= Y + mg cos\theta \sen\phi,\\
        m(\dot{w} + pv - qu) &= Z + mg cos\theta \cos\phi - \kappa T \sen\theta,\\
        I_x \dot{p} + I_{xz} \dot{r} + (I_z - I_y)qr - I_{xz}qp &= L,\\
        I_y \dot{q} + (I_x - I_z)pr + I_{xz}(r^2 - p^2) &= M,\\
        I_z \dot{r} + I_{xz} \dot{p} + (I_y - I_x)qp - I_{xz}qr &= N,
    \end{split}
    \label{eq:boeing747-nonlinear-model_1}
\end{align}
Reescrevemos as equações de movimento \eqref{eq:boeing747-nonlinear-model_1} como,
\begin{align}
    \begin{split}
        m(\dot{u} + qw - rv) &= X - mg\sen\theta + \kappa T \cos\theta,\\
        \dot{u} + qw - rv &= X/m - g\sen\theta + (1/m)\kappa T \cos\theta,\\
        \dot{u} &= - g\sen\theta + (1/m)\kappa T \cos\theta + X/m - qw + rv ,\\
        %
        m(\dot{v} + ru - pw) &= Y + mg cos\theta \sen\phi,\\
        \dot{v} + ru - pw &= Y/m + g cos\theta \sen\phi,\\
        \dot{v} &= g cos\theta \sen\phi + Y/m - ru + pw ,\\
        %
        m(\dot{w} + pv - qu) &= Z + mg cos\theta \cos\phi - \kappa T \sen\theta,\\
        \dot{w} + pv - qu &= Z/m + g cos\theta \cos\phi - (1/m)\kappa T \sen\theta,\\
        \dot{w} &= g cos\theta \cos\phi - (1/m)\kappa T \sen\theta + Z/m  - pv + qu ,\\
        %
        I_x \dot{p} + I_{xz} \dot{r} + (I_z - I_y)qr - I_{xz}qp &= L,\\
        %
        I_y \dot{q} + (I_x - I_z)pr + I_{xz}(r^2 - p^2) &= M,\\
        I_y \dot{q} &= M - (I_x - I_z)pr - I_{xz}(r^2 - p^2) ,\\
        \dot{q} &= M/I_y  - ((I_x - I_z)/I_y) pr - (I_{xz}/I_y) (r^2 - p^2) ,\\
        %
        I_z \dot{r} + I_{xz} \dot{p} + (I_y - I_x)qp - I_{xz}qr &= N,
    \end{split}
    \label{eq:boeing747-nonlinear-model_2}
\end{align}

As equações de movimento translacionais são:
\begin{align}
  \begin{split}
    \dot{u} &= - g\sen\theta + (1/m)\kappa T \cos\theta + X/m - qw + rv,\\
    \dot{v} &= g cos\theta \sen\phi + Y/m - rU + pW ,\\
    \dot{w} &= g cos\theta \cos\phi - (1/m)\kappa T \sen\theta + Z/m  - pv + qu.
  \end{split}
  \label{eq:boeing747-translationsla-dynamics}
\end{align}
Das duas equações a seguir determinamos $\dot{p}$ e $\dot{r}$,
\begin{align}
    \begin{split}
       I_x \dot{p} + I_{xz} \dot{r} + (I_z - I_y)qr + I_{xz}qp &= L,\\
       I_z \dot{r} + I_{xz} \dot{p} + (I_y - I_x)qp - I_{xz}qr &= N,\\
    \end{split}
    \label{eq:pr_acopladas}
\end{align}
Multiplicando a primeira equação por $I_z$ e a segunda por $I_{xz}$, temos:
\begin{align*}
    \begin{split}
        I_zI_x \dot{p} + I_zI_{xz} \dot{r} + I_z(I_z - I_y)qr + I_zI_{xz}qp &= I_zL,\\
        I_{xz}I_z \dot{r} + I_{xz}^2 \dot{p} + I_{xz}(I_y - I_x)qp - I_{xz}^2qr &=I_{xz}N,\\
    \end{split}
\end{align*}
Subtraindo a segunda equação da primeira, temos:
\begin{align*}
         (I_zI_x - I_{xz}^2) \dot{p} + (I_z(I_z - I_y) + I_{xz}^2)qr + (I_zI_{xz} - I_{xz}(I_y - I_x))qp &= I_zL - I_{xz}N,
\end{align*}
Isolando $\dot{p}$, temos:
\begin{align*}
         (I_zI_x - I_{xz}^2) \dot{p} &= I_zL - I_{xz}N - (I_z(I_z - I_y) + I_{xz}^2)qr - (I_zI_{xz} - I_{xz}(I_y - I_x))qp,
\end{align*}
\begin{align*}
         \dot{p} &=\left(\dfrac{1}{I_zI_x - I_{xz}^2}\right)\left( I_zL - I_{xz}N - (I_z(I_z - I_y) + I_{xz}^2)qr - (I_zI_{xz} - I_{xz}(I_y - I_x))qp\right),
\end{align*}
\begin{align*}
    \boxed{\dot{p} =\left(\dfrac{1}{I_zI_x - I_{xz}^2}\right)\left( I_zL - I_{xz}N - (I_z(I_z - I_y) + I_{xz}^2)qr - I_{xz}(I_z -I_y + I_x)qp\right)},
\end{align*}
Para obter \(\dot{r}\), multiplicamos a primeira em equação, \eqref{eq:pr_acopladas}, por \(I_{xz}\) e a segunda por \(I_x\):
\begin{align*}
    \begin{split}
       I_{xz}I_x \dot{p} + I_{xz}^2 \dot{r} + I_{xz}(I_z - I_y)qr + I_{xz}^2 qp &= I_{xz} L,\\
       I_xI_z \dot{r} + I_xI_{xz} \dot{p} + I_x(I_y - I_x) qp - I_xI_{xz} qr &= I_x N,
    \end{split}
\end{align*}
Substituindo a segunda equação na primeira, temos:
\begin{align*}
        (I_{xz}^2-I_xI_z)\dot{r} + (I_{xz}(I_z - I_y) +  I_xI_{xz}) qr + (I_{xz}^2  - I_x(I_y - I_x)) qp &= I_{xz} L - I_x N,
   \end{align*}
\begin{align*}
        (I_{xz}^2-I_xI_z)\dot{r} &= I_{xz} L - I_x N - (I_{xz}(I_z - I_y) +  I_xI_{xz}) qr - (I_{xz}^2  - I_x(I_y - I_x)) qp ,
   \end{align*}
\begin{align*}
    \boxed{\dot{r} = \left(\dfrac{1}{I_{xz}^2-I_xI_z}\right)\left(I_{xz} L - I_x N - (I_{xz}(I_z - I_y) +  I_xI_{xz}) qr - (I_{xz}^2  - I_x(I_y - I_x)) qp\right),}
   \end{align*}
Consequentemente, as equações de movimento podem ser reescritas como:
\begin{align}
    \begin{split}
        \dot{u} &= - g\sen\theta + (1/m)\kappa T \cos\theta - qw + rv + X/m  ,\\
        \dot{v} &= g cos\theta \sen\phi  - ru + pw + Y/m\\
        \dot{w} &= g cos\theta \cos\phi - (1/m)\kappa T \sen\theta  - pv + qu + Z/m ,\\
        \dot{p} & =\left(\dfrac{1}{I_zI_x - I_{xz}^2}\right)\left( I_zL - I_{xz}N - (I_z(I_z - I_y) + I_{xz}^2)qr - I_{xz}(I_z -I_y + I_x)qp\right),\\
        \dot{q} &= M/I_y  - ((I_x - I_z)/I_y) pr - (I_{xz}/I_y) (r^2 - p^2) ,\\
        \dot{r} &= \left(\dfrac{1}{I_{xz}^2-I_xI_z}\right)\left(I_{xz} L - I_x N - (I_{xz}(I_z - I_y) +  I_xI_{xz}) qr - (I_{xz}^2  - I_x(I_y - I_x)) qp\right).
    \end{split}
    \label{eq:boeing747-nonlinear-model_3}
\end{align}
Segundo Bryson \cite{brysonControlSpacecraftAircraft1994}, a relação entre as variações dos ângulos de Euler e as velocidades angulares do corpo é dada por:
\begin{align}
    \begin{split}
        \dot{\phi} &= p + q\sen\phi\tan\theta + r\cos\phi\tan\theta,\\
        \dot{\theta} &= q\cos\phi - r\sen\phi,\\
        \dot{\psi} &= q\sen\phi/\cos\theta + r\cos\phi/\cos\theta.
    \end{split}
    \label{eq:boeing747-euler-angles}
\end{align}

\subsection{Dinâmica completa}
\label{subsection:Boeing747DinTotal}
A  dinâmica total é dada por:

\begin{align}
  \begin{split}
    \dot{u} &= - g\sen\theta + (1/m)\kappa T \cos\theta + X/m - qw + rv,\\
    \dot{v} &= g cos\theta \sen\phi + Y/m - ru + pw,\\
    \dot{w} &= g cos\theta \cos\phi - (1/m)\kappa T \sen\theta + Z/m  - pv + qu,\\
              \dot{\phi} &= p + q\sen\phi\tan\theta + r\cos\phi\tan\theta,\\
        \dot{\theta} &= q\cos\phi - r\sen\phi,\\
        \dot{\psi} &= q\sen\phi/\cos\theta + r\cos\phi/\cos\theta,\\
        \dot{p} & =\left(\dfrac{1}{I_zI_x - I_{xz}^2}\right)\left( I_zL - I_{xz}N - (I_z(I_z - I_y) + I_{xz}^2)qr - I_{xz}(I_z -I_y + I_x)qp\right),\\
        \dot{q} &= M/I_y  - ((I_x - I_z)/I_y) pr - (I_{xz}/I_y) (r^2 - p^2) ,\\
        \dot{r} &= \left(\dfrac{1}{I_{xz}^2-I_xI_z}\right)\left(I_{xz} L - I_x N - (I_{xz}(I_z - I_y) +  I_xI_{xz}) qr - (I_{xz}^2  - I_x(I_y - I_x)) qp\right).
  \end{split}
  \label{eq:Boeing747FullDyn}
\end{align}   

\section{Dinâmica Linear}
\label{sec:DinamicaLinear}

Segundo \cite{franklin2013,brysonControlSpacecraftAircraft1994}, em estado estacionário, em voo rectilineo, temos que \(\dot{u}=\dot{v}=\dot{w}=\dot{p}=\dot{q}=\dot{r}=0\), as asas constumann estar niveladas, \(\phi=0\), consequentemente, da equação \eqref{eq:boeing747-translationsla-dynamics}, temos:

\begin{align}
  \begin{split}
    0 &= - mg\sen\theta + \kappa T \cos\theta + X,\\
    0 &=  Y,\\
    0 &= mg cos\theta - \kappa T \sen\theta + Z,\\
    0 &= L = M = N.    
  \end{split}
  \label{eq:}
\end{align}

meu objetivo é chegar nesse resultado pela expasão de Taylor da dinâmica total.

Seja a função
\begin{align}
  \label{eq:ode-geral}
\dot{y}=f(y,u)
\end{align}
com \(f:\mathbb{R}\times\mathbb{R}\to\mathbb{R}\) e o ponto de interesse \((y_{0},u_{0})\), tal que \((y,u)=(y_{0}+\bar{y}, u_{0}+\bar{u})\), onde \(\bar{y}\) representa uma pequena variação na vizinhança de \(y_{0}\) e \(\bar{u}\) uma pequena variação na vizinhança de \(u_{0}\). A expansão da série de Taylor ao redor de \((y_{0}, u_{0})\) é dada por \cite{CalculoVol22021}:
\begin{align}
  f(y,u)=f(y_{0},u_{0})+\sum_{n=1}^{\infty}\dfrac{1}{n!}\left[(y-y_{0})\dfrac{\partial}{\partial y} + (u-u_{0})\dfrac{\partial}{\partial u}\right]^{n}f(y_{0},u_{0})
  \label{eq:serie-taylor-f-escalar}
\end{align}

Os  primeiros termos da série podem ser escritor na forma matricial na forma:

\begin{align}
  f(y,u) \approx f(y_{0},u_{0}) + \nabla f(y_{0},u_{0})^{T}\mathbf{z} + \dfrac{1}{2} \mathbf{z}^{T}H_{f}(y_{0},u_{0})\mathbf{z}.
  \label{eq:serie-taylor-vetorial}
\end{align}
onde \(\mathbf{z}=(y-y_{0},u-u_{0})^{T}\).

Agora, seja \eqref{eq:ode-geral} com \(f:\mathbb{R}^{n}\times\mathbb{R}^{m}\to\mathbb{R}^{n}\), tal que \(f=(f_{1},\ldots,f_{n})\), onde \(f_{i}:\mathbb{R}^{n}\times\mathbb{R}^{m}\to\mathbb{R}\). Temos que a expansão da série de Taylor para cada \(f_{i}\) é dada por:
\begin{align}
  f_{i}(y,u)=f_{i}(y_{0},u_{0})+\sum_{n=1}^{\infty}\dfrac{1}{n!}\left[\left(\dfrac{\partial }{\partial y}\right)^{T}(y-y_{0}) + \left(\dfrac{\partial}{\partial u}\right)^{T}(u-u_{0})\right]^{n}f_{i}(y_{0},u_{0})
  \label{eq:serie-taylor-fi}
\end{align}
desdobrando os dois primeiros dos termos da série temos:
\begin{align}
  \begin{split}
    f_{i}(y,u)&=f_{i}(y_{0},u_{0})+\left(\dfrac{\partial f_{i}(y_{0},u_{0})}{\partial y}\right)^{T}(y-y_{0}) + \left(\dfrac{\partial f_{i}(y_{0},u_{0})}{\partial u}\right)^{T}(u-u_{0})\\
              & + \dfrac{1}{2}\bigg[(y-y_{0})^{T}\left(\dfrac{\partial^{2} f_{i}(y_{0},u_{0})}{\partial y^{2}}\right)(y-y_{0}) + 2(y-y_{0})^{T}\left(\dfrac{\partial^{2}f_{i}(y_{0},u_{0})}{\partial y\partial u}\right)(u-u_{0})\\
    &+  (u-u_{0})^{T}\left(\dfrac{\partial^{2} f_{i}(y_{0},u_{0})}{\partial u^{2}}\right)(u-u_{0})\bigg] + \ldots
  \end{split}
  \label{eq:serie-taylor-fi-expanded}
\end{align}
Lembrando que \(y=(y_{1},\dots,y_{n})\)$^{T}$ and \(u=(u_{1},\ldots,u_{m})^{T}\). Cada termo na equação \eqref{eq:serie-taylor-fi-expanded} é definido da seguinte forma:
\begin{align}
  \dfrac{\partial f_{i}(y_{0},u_{0})}{\partial y}=\left(\dfrac{\partial f_{i}}{\partial y_{1}},\ldots,\dfrac{\partial f_{i}}{\partial y_{n}}\right)^{T}_{(y_{0},u_{0})}
  \label{eq:jacobian-y}
\end{align}
\begin{align}
  \dfrac{\partial f_{i}(y_{0},u_{0})}{\partial u}=\left(\dfrac{\partial f_{i}}{\partial u_{1}},\ldots,\dfrac{\partial f_{i}}{\partial u_{m}}\right)^{T}_{(y_{0},u_{0})}
  \label{eq:jacobian-u}
\end{align}
\renewcommand{\arraystretch}{2.5}
\begin{align}
  \dfrac{\partial^{2} f_{i}(y_{0},u_{0})}{\partial y^{2}} =
  \begin{bmatrix}
    \dfrac{\partial^{2} f_{i}}{\partial y^{2}_{1}}&\dfrac{\partial^{2} f_{i}}{\partial y_{1}y_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial y_{1}y_{n}}\\
    \dfrac{\partial^{2} f_{i}}{\partial y_{2}y_{1}}&\dfrac{\partial^{2} f_{i}}{\partial y_{2}^{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial y_{2}y_{n}}\\
    \vdots &\vdots&\ddots&\vdots\\
    \dfrac{\partial^{2} f_{i}}{\partial y_{n}y_{1}}&\dfrac{\partial^{2} f_{i}}{\partial y_{n}y_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial y_{n}^{2}}\\
  \end{bmatrix}_{(y_{0},u_{0})}
  \label{eq:hessian-yy}
\end{align}
\begin{align}
  \dfrac{\partial^{2} f_{i}(y_{0},u_{0})}{\partial y\partial u} =
  \begin{bmatrix}
    \dfrac{\partial^{2} f_{i}}{\partial y_{1}u_{1}}&\dfrac{\partial^{2} f_{i}}{\partial y_{1}u_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial y_{1}u_{m}}\\
\dfrac{\partial^{2} f_{i}}{\partial y_{2}u_{1}}&\dfrac{\partial^{2} f_{i}}{\partial y_{2}u_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial y_{2}u_{m}}\\
    \vdots &\vdots&\ddots&\vdots\\
\dfrac{\partial^{2} f_{i}}{\partial y_{n}u_{1}}&\dfrac{\partial^{2} f_{i}}{\partial y_{n}u_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial y_{n}u_{m}}\\
  \end{bmatrix}_{(y_{0},u_{0})}
  \label{eq:hessian-yu}
\end{align}
\begin{align}
  \dfrac{\partial^{2} f_{i}(y_{0},u_{0})}{\partial u^{2}} =
  \begin{bmatrix}
    \dfrac{\partial^{2} f_{i}}{\partial u^{2}_{1}}&\dfrac{\partial^{2} f_{i}}{\partial u_{1}u_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial u_{1}u_{m}}\\
    \dfrac{\partial^{2} f_{i}}{\partial u_{2}u_{1}}&\dfrac{\partial^{2} f_{i}}{\partial u_{2}^{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial u_{2}u_{m}}\\
    \vdots &\vdots&\ddots&\vdots\\
    \dfrac{\partial^{2} f_{i}}{\partial u_{m}u_{1}}&\dfrac{\partial^{2} f_{i}}{\partial u_{m}u_{2}}&\ldots&\dfrac{\partial^{2} f_{i}}{\partial u_{m}^{2}}\\
  \end{bmatrix}_{(y_{0},u_{0})}
  \label{eq:hessian-uu}
\end{align}
